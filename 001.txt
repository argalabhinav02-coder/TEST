function doPost(e) {
  const SHEET_ID = "13himKVCwCc8KtqHKTMNcUcll3ITr6uBPQ4KrXM8V98U";
  const SHEET_NAME = "Users";

  const sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_NAME);
  const data = JSON.parse(e.postData.contents);
  const rows = sheet.getDataRange().getValues();

  for (let i = 1; i < rows.length; i++) {
    const [user, pass, status] = rows[i];

    if (user === data.username && pass === data.password) {
      if (status !== "active") {
        return json({ success: false, message: "Blocked" });
      }
      return json({ success: true });
    }
  }

  return json({ success: false, message: "Invalid" });
}

function json(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
